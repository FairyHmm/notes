$$
\begin{flalign}

& \Big\{ g[f(x)] \Big\}' = f'(x) g'[f(x)] &

\end{flalign}
$$

# Operations

$$
\begin{flalign}

(fg)' &= f'g+fg' &
\\
\left( \dfrac{f}{g} \right)' &= \dfrac{f'g-fg'}{g^{2}}
\\
\sqrt{ x }' &= \dfrac{1}{2 \sqrt{ x }}
\\
|x|' &= \dfrac{x}{|x|}
\\
(f^{-1})' &= \dfrac{1}{f'(f^{-1})}
\\
(f^{\displaystyle g})' &= f^{\displaystyle g} (\ln f + g)'

\end{flalign}
$$

# Trigonometry


$$
\begin{flalign}

&\begin{array}{rlrl}
\sin'(x) &= \cos(x) & \cos'(x) &= -\sin(x)
\\
\tan'(x) &= \dfrac{1}{\cos^2 x} = \sec^2(x) & \cot'(x) &= \dfrac{-1}{\sin^2 x} = -\csc^2(x)
\\
\csc'(x) &= \dfrac{-\cos x}{\sin^2 x} = -\csc(x) \cot(x) & \sec'(x) &= \dfrac{\sin x}{\cos^2 x} = \sec(x) \tan(x)
\\
\arcsin'(x) &= \dfrac{1}{\sqrt{1 - x^2}} & \arccos'(x) &= \dfrac{-1}{\sqrt{1 - x^2}}
\\
\arctan'(x) &= \dfrac{1}{1 + x^2} & \operatorname{arccot}'(x) &= \dfrac{-1}{1 + x^2}
\\
\operatorname{arccsc}'(x) &= \dfrac{-1}{|x| \sqrt{x^2 - 1}} & \operatorname{arcsec}'(x) &= \dfrac{1}{|x| \sqrt{x^2 - 1}}
\end{array} &

\end{flalign}
$$

# Higher derivative

$$
\begin{flalign}

(f g)^{(n)}  &= \sum_{r=0}^{n} \binom{n}{k} f^{(k)} g^{(n-k)}
\\
\left[ (1-x)^{-k} \right]^{(n)}  &= \dfrac{n! \dfrac{(k+n-1)!}{(k-1)!}}{(1-x)^{k+n}} &
\\
\dfrac{1}{x-a}^{(n)} &= \dfrac{(-1)^n n!}{(x - a)^{n+1}}

\end{flalign}
$$
